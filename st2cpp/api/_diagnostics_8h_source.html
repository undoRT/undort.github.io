166struct SourceLocation {
167 std::string fileName;
168 uint32_t line = 0;
169 uint32_t column = 0;
170 uint32_t endLine = 0;
171 uint32_t endColumn = 0;
172
173 bool isValid() const { return line != 0; }
174
175 std::string toString() const {
176 if (!isValid()) return "<unknown>";
177 std::string result = fileName;
178 result += ':' + std::to_string(line);
179 result += ':' + std::to_string(column);
180 return result;
181 }

187struct Diagnostic {
188 DiagnosticSeverity severity;
189 DiagnosticCode code;
190 std::string message;
191 SourceLocation location;
192 std::vector<SourceLocation> relatedLocations; // e.g., "see declaration here"
193 std::string suggestion; // Optional fix hint
194
195 std::string toString() const {
196 std::string sev;
197 switch (severity) {
198 case DiagnosticSeverity::Error: sev = "error"; break;
199 case DiagnosticSeverity::Warning: sev = "warning"; break;
200 case DiagnosticSeverity::Note: sev = "note"; break;
201 }
202 std::string result = location.toString() + ": " + sev + ": [" + diagnosticCodeToString(code) + "] " + message;
203 if (!suggestion.empty()) {
204 result += " (suggestion: " + suggestion + ")";
205 }
206 return result;
207 }
208};

267 // Print formatted output (GCC/Clang style)

269 for (const auto& d : diagnostics_) {
270 out << d.toString() << '\n';
