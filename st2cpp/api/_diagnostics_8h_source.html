172struct SourceLocation {
173 std::string fileName;
174 uint32_t line = 0;
175 uint32_t column = 0;
176 uint32_t endLine = 0;
177 uint32_t endColumn = 0;
178
179 bool isValid() const { return line != 0; }
180
181 std::string toString() const {
182 if (!isValid()) return "<unknown>";
183 std::string result = fileName;
184 result += ':' + std::to_string(line);
185 result += ':' + std::to_string(column);
186 return result;
187 }

193struct Diagnostic {
194 DiagnosticSeverity severity;
195 DiagnosticCode code;
196 std::string message;
197 SourceLocation location;
198 std::vector<SourceLocation> relatedLocations; // e.g., "see declaration here"
199 std::string suggestion; // Optional fix hint
200
201 std::string toString() const {
202 std::string sev;
203 switch (severity) {
204 case DiagnosticSeverity::Error: sev = "error"; break;
205 case DiagnosticSeverity::Warning: sev = "warning"; break;
206 case DiagnosticSeverity::Note: sev = "note"; break;
207 }
208 std::string result = location.toString() + ": " + sev + ": [" + diagnosticCodeToString(code) + "] " + message;
209 if (!suggestion.empty()) {
210 result += " (suggestion: " + suggestion + ")";
211 }
212 return result;
213 }

317 // Extract the raw text of a 1-based source line.
318 static std::string sourceLine(const std::string& text, uint32_t line) {
319 if (text.empty() || line == 0) return "";
320 size_t start = 0;
321 for (uint32_t i = 1; i < line; ++i) {
322 size_t nl = text.find('\n', start);
323 if (nl == std::string::npos) return "";
324 start = nl + 1;
325 }
326 size_t end = text.find('\n', start);
327 std::string l = text.substr(start, end == std::string::npos ? std::string::npos : end - start);
328 if (!l.empty() && l.back() == '\r') l.pop_back();
329 return l;
330 }

332 // Expand tabs to spaces so the source line and its caret stay aligned.
333 static std::string expandTabs(const std::string& src) {
334 std::string out;
335 size_t disp = 0;
336 for (char c : src) {
337 if (c == '\t') {
338 const size_t spaces = 4 - (disp % 4);
339 out.append(spaces, ' ');
340 disp += spaces;
341 } else {
342 out += c;
343 ++disp;
344 }
345 }
346 return out;
347 }

349 // Build the caret underline for a 1-based column (optionally spanning to
350 // endColumn). Tabs before the column are taken into account.
351 static std::string caretLine(const std::string& src, uint32_t col, uint32_t endColumn) {
352 if (col == 0) return "";
353 size_t disp = 0; // display column of `col` in the expanded line
354 size_t idx = 0;
355 while (idx < src.size() && idx + 1 < static_cast<size_t>(col)) {
356 if (src[idx] == '\t') {
357 disp = (disp / 4 + 1) * 4;
358 } else {
359 ++disp;
360 }
361 ++idx;
362 }
363 size_t span = 1;
364 if (endColumn > col) {
365 span = static_cast<size_t>(endColumn) - col;
366 }
367 return std::string(disp, ' ') + std::string(span, '^');
368 }

370 void printDiagnostic(std::ostream& out, const Diagnostic& d) const {
371 // Header line: file:line:col: severity: message [Code]
372 if (d.location.isValid()) {
373 out << d.location.toString() << ": ";
374 } else {
375 std::string file = d.location.fileName.empty() ? sourceName_ : d.location.fileName;
376 out << file << ": ";
377 }
378 out << severityName(d.severity) << ": " << d.message
379 << " [" << diagnosticCodeToString(d.code) << "]\n";
380
381 // Code snippet with a caret (only when a source position and the
382 // analyzed source text are available).
383 if (d.location.isValid() && d.location.column > 0 && !sourceText_.empty()) {
384 const std::string line = expandTabs(sourceLine(sourceText_, d.location.line));
385 if (!line.empty()) {
386 const std::string num = std::to_string(d.location.line);
387 const std::string pad(num.size(), ' ');
388 const std::string caret
389 = caretLine(line, d.location.column, d.location.endColumn);
390 out << " " << num << " | " << line << '\n';
391 out << " " << pad << " | " << caret << '\n';
392 }
393 }
394 for (const auto& rel : d.relatedLocations) {
395 out << rel.toString() << ": note: see here\n";
396 }
397 if (!d.suggestion.empty()) {
398 out << " suggestion: " << d.suggestion << '\n';
399 }
400 }
