392 std::vector<SymbolId> effectiveParams(SymbolId fbId) const {
393 // Base chain, root ancestor first.
394 std::vector<const Symbol*> chain;
395 std::unordered_set<SymbolId> visited;
396 for (const Symbol* cursor = get(fbId);
397 cursor != nullptr && visited.insert(cursor->id).second;
398 cursor = (cursor->baseClassId != 0) ? get(cursor->baseClassId) : nullptr) {
399 chain.push_back(cursor);
400 }
401 std::reverse(chain.begin(), chain.end());
402
403 // First pass, most derived block first: the winning declaration of each
404 // name is the one in the nearest block that redeclares it.
405 std::unordered_map<std::string, SymbolId> winner;
406 for (auto it = chain.rbegin(); it != chain.rend(); ++it) {
407 for (SymbolId paramId : (*it)->params) {
408 const Symbol* param = get(paramId);
409 if (param == nullptr) continue;
410 const std::string key = normalizeKey(param->name);
411 if (winner.count(key) != 0) continue; // a derived block already declared it
412 winner.emplace(key, paramId);
413 }
414 }
415
416 // Second pass, base first, so the base interface stays the prefix that
417 // positional calls bind against. Each name is emitted once.
418 std::vector<SymbolId> out;
419 std::unordered_set<std::string> emitted;
420 for (const Symbol* fb : chain) {
421 for (SymbolId paramId : fb->params) {
422 const Symbol* param = get(paramId);
423 if (param == nullptr) continue;
424 const std::string key = normalizeKey(param->name);
425 // Test the winner first: a shadowed declaration must not mark
426 // the name as taken, or the winning one below would be dropped.
427 auto win = winner.find(key);
428 if (win == winner.end() || win->second != paramId) continue;
429 if (!emitted.insert(key).second) continue;
430 out.push_back(paramId);
431 }
432 }
433 return out;
434 }
