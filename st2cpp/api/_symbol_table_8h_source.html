168 SymbolId findInScope(const Scope* scope, const std::string& name) const
169 {
170 if (!scope) {
171 return 0;
172 }
173 auto it = scope->symbols.find(normalizeKey(name));
174 if (it != scope->symbols.end()) {
175 return it->second;
176 }
177 if (caseSensitive_ && caseFallback_) {
178 // In case-sensitive mode a declaration is stored under its verbatim
179 // spelling, so the miss cannot be retried with a second key: `Name`
180 // may have been declared `name` or `NAME`. Scan the scope instead.
181 const std::string upper = asciiUpper(name);
182 if (upper != name) {
183 SymbolId found = 0;
184 for (const auto& entry : scope->symbols) {
185 if (asciiUpper(entry.first) != upper) {
186 continue;
187 }
188 if (found != 0) {
189 // Two declarations differ only by case: falling back
190 // would pick one arbitrarily, so report neither and let
191 // the reference fail rather than guess.
192 return 0;
193 }
194 found = entry.second;
195 }
196 if (found != 0) {
197 recordCaseMismatch(name, found);
198 return found;
199 }
200 }
201 }
202 return 0;
203 }

514 std::vector<SymbolId> effectiveParams(SymbolId fbId) const
515 {
516 // Base chain, root ancestor first.
517 std::vector<const Symbol*> chain;
518 std::unordered_set<SymbolId> visited;
519 for (const Symbol* cursor = get(fbId); cursor != nullptr && visited.insert(cursor->id).second;
520 cursor = (cursor->baseClassId != 0) ? get(cursor->baseClassId) : nullptr) {
521 chain.push_back(cursor);
522 }
523 std::reverse(chain.begin(), chain.end());
524
525 // First pass, most derived block first: the winning declaration of each
526 // name is the one in the nearest block that redeclares it.
527 std::unordered_map<std::string, SymbolId> winner;
528 for (auto it = chain.rbegin(); it != chain.rend(); ++it) {
529 for (SymbolId paramId : (*it)->params) {
530 const Symbol* param = get(paramId);
531 if (param == nullptr) {
532 continue;
533 }
534 const std::string key = normalizeKey(param->name);
535 if (winner.count(key) != 0) {
536 continue; // a derived block already declared it
537 }
538 winner.emplace(key, paramId);
539 }
540 }
541
542 // Second pass, base first, so the base interface stays the prefix that
543 // positional calls bind against. Each name is emitted once.
544 std::vector<SymbolId> out;
545 std::unordered_set<std::string> emitted;
546 for (const Symbol* fb : chain) {
547 for (SymbolId paramId : fb->params) {
548 const Symbol* param = get(paramId);
549 if (param == nullptr) {
550 continue;
551 }
552 const std::string key = normalizeKey(param->name);
553 // Test the winner first: a shadowed declaration must not mark
554 // the name as taken, or the winning one below would be dropped.
555 auto win = winner.find(key);
556 if (win == winner.end() || win->second != paramId) {
557 continue;
558 }
559 if (!emitted.insert(key).second) {
560 continue;
561 }
562 out.push_back(paramId);
563 }
564 }
565 return out;
566 }
