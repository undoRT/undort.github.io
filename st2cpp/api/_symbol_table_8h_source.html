212 SymbolId lookupRecursive(const std::string& name) const {
213 std::string key = normalizeKey(name);
214 ScopeId scopeId = currentScopeId_;
215 while (scopeId != 0) {
216 const Scope* scope = getScope(scopeId);
217 if (!scope) break;
218 auto it = scope->symbols.find(key);
219 if (it != scope->symbols.end()) {
220 return it->second;
221 }
222 scopeId = scope->parentId;
