302 std::string key = normalizeKey(name);
303 ScopeId scopeId = currentScopeId_;
304 while (scopeId != 0) {
305 const Scope* scope = getScope(scopeId);
306 if (!scope) break;
307 auto it = scope->symbols.find(key);
