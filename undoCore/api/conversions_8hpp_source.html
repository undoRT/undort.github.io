131/*
132 * BCD conversions (Binary-Coded Decimal)
133 * Common in industrial displays, rotary encoders, thumbwheel switches.
134 * Each nibble (4 bits) encodes one decimal digit [0..9].
135 */
136
141inline USINT BCD_TO_USINT(BYTE bcd)
142{
143 UInt8 lo = bcd & 0x0F;
144 UInt8 hi = (bcd >> 4) & 0x0F;
145 if (lo > 9 || hi > 9) {
146 throw std::invalid_argument("BCD_TO_USINT: invalid BCD digit");
147 }
148 return static_cast<USINT>(hi * 10 + lo);
149}
150
155inline BYTE USINT_TO_BCD(USINT value)
156{
157 if (value > 99) {
158 throw std::out_of_range("USINT_TO_BCD: value exceeds 2-digit BCD range");
159 }
160 return static_cast<BYTE>(((value / 10) << 4) | (value % 10));
161}
162
167inline UINT BCD_TO_INT(WORD bcd)
168{
169 UInt16 result = 0;
170 UInt16 multiplier = 1;
171 for (int shift = 0; shift < 16; shift += 4) {
172 UInt16 nibble = (bcd >> shift) & 0x0F;
173 if (nibble > 9) {
174 throw std::invalid_argument("BCD_TO_INT: invalid BCD digit");
175 }
176 result += static_cast<UInt16>(nibble * multiplier);
177 multiplier *= 10;
178 }
179 return static_cast<UINT>(result);
180}
181
186inline WORD INT_TO_BCD(UINT value)
187{
188 if (value > 9999) {
189 throw std::out_of_range("INT_TO_BCD: value exceeds 4-digit BCD range");
190 }
191 UInt16 result = 0;
192 UInt16 v = value;
193 for (int shift = 0; shift < 16 && v > 0; shift += 4) {
194 result |= static_cast<UInt16>((v % 10) << shift);
195 v /= 10;
196 }
197 return static_cast<WORD>(result);
198}

238/*
239 * STRING <-> WSTRING (width conversion)
240 *
241 * NOTE: this is a minimal, locale-independent conversion intended for
242 * ASCII/Latin-1 industrial text (tag names, alarm messages). It does not
243 * perform full Unicode-aware transcoding. Revisit if multi-byte UTF-8
244 * support becomes a requirement for HMI text.
245 */
246
251inline STRING WSTRING_TO_STRING(const WSTRING& ws)
252{
253 STRING result;
254 result.reserve(ws.size());
255 for (wchar_t wc : ws) {
256 if (static_cast<unsigned long>(wc) > 0xFF) {
257 throw std::range_error("WSTRING_TO_STRING: character out of Latin-1 range");
258 }
259 result.push_back(static_cast<char>(wc));
260 }
261 return result;
262}

283/*
284 * DATE / TIME_OF_DAY / DATE_AND_TIME conversions
285 *
286 * Representation recap (see types.hpp):
287 * DATE -> seconds since 1970-01-01 UTC, truncated to midnight
288 * TIME_OF_DAY(TOD)-> milliseconds since midnight
289 * DATE_AND_TIME(DT)-> seconds since 1970-01-01 UTC (full timestamp)
290 *
291 * All calendar decomposition uses gmtime (UTC), never local time.
292 */
293
297inline DATE DT_TO_DATE(DT dt)
298{
299 std::time_t t = static_cast<std::time_t>(dt);
300 std::tm tmVal{};
301 gmtime_r(&t, &tmVal);
302 tmVal.tm_hour = 0;
303 tmVal.tm_min = 0;
304 tmVal.tm_sec = 0;
305 return static_cast<DATE>(timegm(&tmVal));
306}
307
311inline TOD DT_TO_TOD(DT dt)
312{
313 std::time_t t = static_cast<std::time_t>(dt);
314 std::tm tmVal{};
315 gmtime_r(&t, &tmVal);
316 UInt64 msOfDay = (static_cast<UInt64>(tmVal.tm_hour) * 3600ULL
317 + static_cast<UInt64>(tmVal.tm_min) * 60ULL
318 + static_cast<UInt64>(tmVal.tm_sec)) * 1000ULL;
319 return static_cast<TOD>(msOfDay);
320}

366inline STRING TOD_TO_STRING(TOD tod)
367{
368 UInt64 totalMs = tod;
369 UInt32 h = static_cast<UInt32>(totalMs / 3600000ULL);
370 UInt32 m = static_cast<UInt32>((totalMs / 60000ULL) % 60ULL);
371 UInt32 s = static_cast<UInt32>((totalMs / 1000ULL) % 60ULL);
372 char buf[16];
373 std::snprintf(buf, sizeof(buf), "%02u:%02u:%02u", h, m, s);
374 return STRING(buf);
375}
