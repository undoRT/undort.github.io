1409inline USINT BCD_TO_USINT(BYTE bcd)
1410{
1411 UInt8 lo = bcd & 0x0F;
1412 UInt8 hi = (bcd >> 4) & 0x0F;
1413 if (lo > 9 || hi > 9) {
1414 throw std::invalid_argument("BCD_TO_USINT: invalid BCD digit");
1415 }
1416 return static_cast<USINT>(hi * 10 + lo);
1417}
1418
1423inline BYTE USINT_TO_BCD(USINT value)
1424{
1425 if (value > 99) {
1426 throw std::out_of_range("USINT_TO_BCD: value exceeds 2-digit BCD range");
1427 }
1428 return static_cast<BYTE>(((value / 10) << 4) | (value % 10));
1429}
1430
1435inline UINT BCD_TO_INT(WORD bcd)
1436{
1437 UInt16 result = 0;
1438 UInt16 multiplier = 1;
1439 for (int shift = 0; shift < 16; shift += 4) {
1440 UInt16 nibble = (bcd >> shift) & 0x0F;
1441 if (nibble > 9) {
1442 throw std::invalid_argument("BCD_TO_INT: invalid BCD digit");
1443 }
1444 result += static_cast<UInt16>(nibble * multiplier);
1445 multiplier *= 10;
1446 }
1447 return static_cast<UINT>(result);
1448}
1449
1454inline WORD INT_TO_BCD(UINT value)
1455{
1456 if (value > 9999) {
1457 throw std::out_of_range("INT_TO_BCD: value exceeds 4-digit BCD range");
1458 }
1459 UInt16 result = 0;
1460 UInt16 v = value;
1461 for (int shift = 0; shift < 16 && v > 0; shift += 4) {
1462 result |= static_cast<UInt16>((v % 10) << shift);
1463 v /= 10;
1464 }
1465 return static_cast<WORD>(result);
1466}

1545/*
1546 * DATE / TIME_OF_DAY / DATE_AND_TIME conversions
1547 *
1548 * Representation recap (see types.hpp):
1549 * DATE -> seconds since 1970-01-01 UTC, truncated to midnight
1550 * TIME_OF_DAY(TOD)-> milliseconds since midnight
1551 * DATE_AND_TIME(DT)-> seconds since 1970-01-01 UTC (full timestamp)
1552 *
1553 * All calendar decomposition uses gmtime (UTC), never local time.
1554 */
1555
1559inline DATE DT_TO_DATE(DT dt)
1560{
1561 std::time_t t = static_cast<std::time_t>(dt);
1562 std::tm tmVal{};
1563 gmtime_r(&t, &tmVal);
1564 tmVal.tm_hour = 0;
1565 tmVal.tm_min = 0;
1566 tmVal.tm_sec = 0;
1567 return static_cast<DATE>(timegm(&tmVal));
1568}
1569
1573inline TOD DT_TO_TOD(DT dt)
1574{
1575 std::time_t t = static_cast<std::time_t>(dt);
1576 std::tm tmVal{};
1577 gmtime_r(&t, &tmVal);
1578 UInt64 msOfDay = (static_cast<UInt64>(tmVal.tm_hour) * 3600ULL + static_cast<UInt64>(tmVal.tm_min) * 60ULL
1579 + static_cast<UInt64>(tmVal.tm_sec))
1580 * 1000ULL;
1581 return static_cast<TOD>(msOfDay);
1582}
