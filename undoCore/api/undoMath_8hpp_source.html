478template<typename T>
479inline T SHL(T x, int n)
480{
481 static_assert(detail::is_bitstring<T>::value, "SHL is only defined for bit string types (BYTE, WORD, DWORD, LWORD)");
482 constexpr int width = detail::bit_width<T>::value;
483 n = n % width;
484 return static_cast<T>((static_cast<typename std::make_unsigned<T>::type>(x) << n) & detail::allBits<T>());
485}

533// ============================================================================
534// Rotate left
535// ============================================================================
536
545template<typename T>
546inline T ROL(T x, int n)
547{
548 static_assert(detail::is_bitstring<T>::value, "ROL is only defined for bit string types (BYTE, WORD, DWORD, LWORD)");
549 constexpr int width = detail::bit_width<T>::value;
550 n = n % width;
551 using U = typename std::make_unsigned<T>::type;
552 U ux = static_cast<U>(x);
553 U result = (ux << n) | (ux >> (width - n));
554 return static_cast<T>(result & detail::allBits<T>());
555}

811template<typename T>
812inline T DIV(T x, T y)
813{
814 static_assert(std::is_integral<T>::value, "DIV is only defined for integral types");
815 if constexpr (std::is_signed<T>::value) {
816 return static_cast<T>(std::trunc(static_cast<LREAL>(x) / static_cast<LREAL>(y)));
817 } else {
818 return static_cast<T>(x / y);
819 }
820}

829template<typename T>
830inline T MOD(T x, T y)
831{
832 static_assert(std::is_integral<T>::value, "MOD is only defined for integral types");
833 if constexpr (std::is_signed<T>::value) {
834 // For signed integers, result has same sign as x
835 T quotient = static_cast<T>(std::trunc(static_cast<LREAL>(x) / static_cast<LREAL>(y)));
836 return static_cast<T>(x - quotient * y);
837 } else {
838 return static_cast<T>(x % y);
839 }
840}
