297 const std::string* cycle(int direction)
298 {
299 if (m_documents.empty()) {
300 return nullptr;
301 }
302 int index = -1;
303 for (size_t i = 0; i < m_documents.size(); ++i) {
304 if (m_documents[i].path == m_active) {
305 index = static_cast<int>(i);
306 break;
307 }
308 }
309 const int count = static_cast<int>(m_documents.size());
310 // With nothing active, "next" starts at the beginning and "previous" at the
311 // end, so either key goes somewhere rather than jumping to the first twice.
312 int next = (index < 0) ? (direction >= 0 ? 0 : count - 1) : (index + direction + count) % count;
313 m_active = m_documents[static_cast<size_t>(next)].path;
