170inline bool isCountSummary(const std::string& line)
171{
172 for (size_t i = 0; i < line.size(); ++i) {
173 if (!std::isdigit(static_cast<unsigned char>(line[i]))) {
174 continue;
175 }
176 size_t after = i;
177 while (after < line.size() && std::isdigit(static_cast<unsigned char>(line[after]))) {
178 ++after;
179 }
180 size_t word = after;
181 while (word < line.size() && line[word] == ' ') {
182 ++word;
183 }
184 const std::string tail = detail::toLowerText(line.substr(word, 12));
185 if (detail::startsWith(tail, "error") || detail::startsWith(tail, "warning")) {
186 return true;
187 }
188 i = after - 1;
189 }
190 return false;
191}

205inline OutSeverity classifyCompilerLine(const std::string& line)
206{
207 if (isCountSummary(line)) {
208 return OutSeverity::Info;
209 }
210
211 const std::string lower = detail::toLowerText(line);
212
213 // "file.st:12:5: error: message", or a line that opens with the word.
214 for (const char* marker : {": error", ": warning", "error:", "warning:"}) {
215 if (lower.find(marker) != std::string::npos) {
216 return (marker[0] == 'e' || marker == std::string(": error")) ? OutSeverity::Error
217 : OutSeverity::Warning;
218 }
219 }
220
221 return OutSeverity::Info;
222}
